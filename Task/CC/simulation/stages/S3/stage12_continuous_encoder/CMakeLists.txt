cmake_minimum_required(VERSION 3.20)
project(stage12_continuous_encoder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../..")
set(CC_ROOT "${REPO_ROOT}/Task/CC")
add_executable(stage12_tests tests/test_stage12.cpp src/continuous_encoder.cpp "${CC_ROOT}/shared/src/trellis.cpp" "${CC_ROOT}/shared/src/puncturing.cpp" "${CC_ROOT}/block/current/src/block_encoder.cpp")
target_include_directories(stage12_tests PRIVATE include "${CC_ROOT}/shared/include" "${CC_ROOT}/block/current/include" "${REPO_ROOT}/Task/Common/include")
if(NOT MSVC)
 target_compile_options(stage12_tests PRIVATE -O3 -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME stage12_continuous_encoder COMMAND stage12_tests "${CMAKE_CURRENT_LIST_DIR}/results")
